 struct bucket_array_void
 {
  Arena *arena;
  bucket_array_bucket *first_bucket;
  bucket_array_bucket *last_bucket;
  i32 bucket_count;
  i32 bucket_cap;
  i32 item_size;
  //-
  
  inline i32 item_count()
  {
   i32 result = 0;
   if (bucket_count > 1) {
    result += (bucket_count-1)*bucket_cap;
   }
   result += last_bucket->count;
   return result;
  }
  
  inline i32 item_cap() {
   return bucket_count * bucket_cap;
  }
  
  void set_count(i32 new_count)
  {
   kv_assert(new_count >= 0);
   i32 cur_count = item_count();
   i32 delta = new_count - cur_count;
   if ( new_count < cur_count )
   {// NOTE(kv): Popping items

    for_i32(auto bucket = last_bucket;
            bucket;
            bucket = bucket->prev)
    {
     if (bucket->count < size_to_pop) {
      size_to_pop -= bucket->count;
      bucket->count = 0;
     } else {
      bucket->count -= size_to_pop;
      size_to_pop = 0;  // @decor
      break;
     }
    }
    kv_assert(size_to_pop == 0);
   }
   else if (new_count > cur_count)
   {// NOTE(kv): Expanding
    i32 cur_cap = get_cap();
    if (new_count <= cur_cap) {
     // NOTE(kv): Lucky, we're within capacity
     last_bucket->count += delta;
     kv_assert(last_bucket->count <= last_bucket->cap);
    } else {
     // NOTE(kv): Need a new bucket
     delta -= last_bucket->cap - last_bucket->count;
     last_bucket->count = last_bucket->cap;

      *new_bucket = {
       .prev  = last_bucket,
       .count = delta,
       .cap   = new_bucket_cap,
      };
      last_bucket->next = new_bucket;
     }
    }
   }
  }
  
  inline void pop() { set_count(count-1); }
  
  void push(T const& item)

 allocated_size = (wanted_size +
                   win32_memory_header_size +
                   /*redzone at the end*/win32_redzone_size);
 allocated_size = (allocated_size+granularity_mask) & (~granularity_mask);
 
 void *base = 0;  //NOTE(kv) We don't control the base address yet!
 void *memory0 = VirtualAlloc(base, (SIZE_T)allocated_size,
                              MEM_COMMIT|MEM_RESERVE, PAGE_READWRITE);
 
 u8 *memory = (u8 *)memory0;
 usize usable_size = allocated_size-(win32_memory_header_size+win32_redzone_size);
 u8 *usable_base = memory + win32_memory_header_size;
 {
  Memory_Annotation_Tracker_Node *node = cast(Memory_Annotation_Tracker_Node *)memory;
  {// TODO(kv): When we use malloc-based allocator, we don't do any of this... :>
   EnterCriticalSection(&memory_tracker_mutex);
   {//NOTE(kv) asan poisoning
    usize tracker_size = sizeof(Memory_Annotation_Tracker_Node);
    //NOTE(kv) The left redzone is smaller since we still needs to access the tracker.
    //  But I don't really care, underflow errors are super rare anyway.
    ASAN_POISON_MEMORY_REGION(memory, allocated_size);
    ASAN_UNPOISON_MEMORY_REGION(memory, tracker_size);
    ASAN_UNPOISON_MEMORY_REGION(usable_base, usable_size);
   }
   zdll_push_back(memory_tracker.first, memory_tracker.last, node);
   memory_tracker.count += 1;
   LeaveCriticalSection(&memory_tracker_mutex);
  }
  node->location = location;
  node->size = allocated_size;
 }
 *usable_size_out = usable_size;
 return usable_base;
}

 u8 *result = (u8 *)result0;
 //NOTE(kv) We don't use the extraneous memory, so poison it!
 kv_assert(usable_size >= size);
 ASAN_POISON_MEMORY_REGION(result+size, usable_size-size);
 return result;
}

  zdll_remove(memory_tracker.first, memory_tracker.last, tracker);
  memory_tracker.count -= 1;
  LeaveCriticalSection(&memory_tracker_mutex);
 }
 VirtualFree(tracker, 0, MEM_RELEASE);
 ASAN_POISON_MEMORY_REGION(tracker, allocated_size);

  //-Find non-conflicting arena

   for_i32(conflict_index,0,conflict_count){
